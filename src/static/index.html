B=http://<IPG-IP>:11524

# decide (preset fills in the questions)
curl -s $B/api/decide -H 'Content-Type: application/json' -d '{
  "model": "laya", "preset": "triage",
  "state": {"message": "You charged me twice, refund NOW!"}
}' | python3 -m json.tool

# decide with your own questions
curl -s $B/api/decide -H 'Content-Type: application/json' -d '{
  "model": "decider:4b", "keep_alive": "10m",
  "state": "I cannot log in since this morning, demo at 3pm.",
  "questions": {"angry": {"type": "noul", "instructions": "Is the customer angry?"}}
}' | python3 -m json.tool

# models
curl -s $B/api/models | python3 -m json.tool            # local models (+default_model)
curl -s $B/api/models/running | python3 -m json.tool   # loaded models
curl -s $B/api/policy | python3 -m json.tool           # loaded-model policy
curl -s -X POST $B/api/policy -d '{"default_model":"laya:latest","max_loaded_models":2}'
curl -s -X POST $B/api/models/show -d '{"model":"laya"}'
curl -s -X POST $B/api/models/load -d '{"model":"laya:en","keep_alive":-1}'
curl -s -X POST $B/api/models/unload -d '{"model":"laya:en"}'
curl -s -X POST $B/api/models/pull -d '{"model":"decider"}'
curl -s -X DELETE $B/api/models -d '{"model":"my-old-model"}'
curl -s -X POST $B/api/models/copy -d '{"source":"laya:en","destination":"my-guard"}'
curl -s -X POST $B/api/models/create -d '{"model":"triage","from":"laya:en"}'

# MCP lifecycle + tools
curl -s -X POST $B/api/mcp/start | python3 -m json.tool
curl -s $B/api/mcp/status | python3 -m json.tool
curl -s -X POST $B/api/mcp/call -H 'Content-Type: application/json' -d '{
  "tool": "decide", "args": {"model": "laya", "preset": "triage",
    "state": "Refund my double charge please"}}' | python3 -m json.tool
curl -s -X POST $B/api/mcp/stop

# generic passthrough to any ollaya path (logged)
curl -s -X POST $B/api/proxy -d '{"method":"GET","path":"/api/tags"}'
curl -s -X POST $B/api/proxy -d '{"method":"POST","path":"/v1/systemone","body":{"model":"laya",...}}'

# auth: login returns a session cookie; API clients use a bearer key instead
curl -s -c jar $B/api/auth/login -H 'Content-Type: application/json' -d '{"username":"admin","password":"..."}'
curl -s -b jar $B/api/auth/me | python3 -m json.tool
curl -s -b jar -X POST $B/api/keys -d '{"name":"n8n"}'   # key shown ONCE
KEY=owi_...; curl -s $B/api/decide -H "Authorization: Bearer $KEY" -d '{"model":"laya",...}'

# users (admin) / profile
curl -s -b jar $B/api/users | python3 -m json.tool
curl -s -b jar -X POST $B/api/users -d '{"username":"ops","password":"...","role":"user"}'

# history + stats (scoped to caller unless admin)
curl -s "$B/api/history?limit=30" | python3 -m json.tool
curl -s $B/api/history/123 | python3 -m json.tool
curl -s $B/api/metrics | python3 -m json.tool
